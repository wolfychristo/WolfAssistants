```javascript
// Chrome Extension: content_script.js
function scrapeLinkedInProfile() {
  const profileData = {
    name: document.querySelector('.text-heading-xlarge')?.innerText,
    title: document.querySelector('.text-body-medium')?.innerText,
    company: document.querySelector('[data-field="experience"] .t-bold span')?.innerText,
    location: document.querySelector('.text-body-small.inline')?.innerText,
    profileUrl: window.location.href,
    photoUrl: document.querySelector('.pv-top-card-profile-picture__image')?.src,
    
    // Recent posts
    recentPosts: Array.from(document.querySelectorAll('.feed-shared-update-v2'))
      .slice(0, 3)
      .map(post => ({
        text: post.querySelector('.feed-shared-text')?.innerText,
        timestamp: post.querySelector('.feed-shared-actor__sub-description')?.innerText
      })),
    
    // Mutual connections
    mutualConnections: document.querySelector('.dist-value')?.innerText,
    
    // Company LinkedIn
    companyLinkedIn: document.querySelector('[data-field="experience"] a')?.href
  };
  
  // Send to backend
  fetch('https://api.wolfassistants.com/contacts/scrape', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${getAuthToken()}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(profileData)
  })
  .then(response => response.json())
  .then(data => {
    showNotification('✓ Contact saved! Researching...');
    // Trigger AI research
    startAIResearch(data.contactId);
  })
  .catch(error => {
    showNotification('❌ Error saving contact', 'error');
  });
}

// Keyboard shortcut: Alt+W
document.addEventListener('keydown', (e) => {
  if (e.altKey && e.key === 'w') {
    scrapeLinkedInProfile();
  }
});
```
